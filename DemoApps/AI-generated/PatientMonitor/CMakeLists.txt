file(GLOB SRCS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/common/application/*.c" Src/*.c)
file(GLOB HDRS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/common/application/*.h" Hdr/*.h)
add_executable(DemoApp_PatientMonitor ${SRCS} ${HDRS})
target_include_directories(DemoApp_PatientMonitor PRIVATE Hdr "${CMAKE_SOURCE_DIR}/common/application")
target_link_libraries(DemoApp_PatientMonitor PRIVATE eve_hal)
if(WIN32)
  target_link_libraries(DemoApp_PatientMonitor PRIVATE user32 gdi32)
endif()
set_property(TARGET DemoApp_PatientMonitor PROPERTY C_STANDARD 11)
if(IS_PLATFORM_RP2040)
  pico_enable_stdio_uart(DemoApp_PatientMonitor 1)
  pico_enable_stdio_usb(DemoApp_PatientMonitor 1)
  pico_add_extra_outputs(DemoApp_PatientMonitor)
endif()
set(PM_DEPLOY "${CMAKE_BINARY_DIR}/deploy/DemoApp_PatientMonitor")
add_custom_command(TARGET DemoApp_PatientMonitor POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E make_directory "${PM_DEPLOY}"
  COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:DemoApp_PatientMonitor>" "${PM_DEPLOY}")
if(IS_PLATFORM_RP2040)
  add_custom_command(TARGET DemoApp_PatientMonitor POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "$<TARGET_FILE_DIR:DemoApp_PatientMonitor>/DemoApp_PatientMonitor.uf2" "${PM_DEPLOY}/DemoApp_PatientMonitor.uf2")
endif()
if(WIN32)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(PM_ARCH x64)
  else()
    set(PM_ARCH x86)
  endif()
  file(GLOB PM_DLLS "${CMAKE_SOURCE_DIR}/common/eve_hal/Bin/MSVC/${PM_ARCH}/*.dll"
    "${CMAKE_SOURCE_DIR}/common/eve_hal/Bin/Simulation/${PM_ARCH}/*.dll")
  foreach(PM_DLL IN LISTS PM_DLLS)
    add_custom_command(TARGET DemoApp_PatientMonitor POST_BUILD
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${PM_DLL}" "${PM_DEPLOY}")
  endforeach()
endif()
